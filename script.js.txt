function switchTab(event, tabId) {
    document.querySelectorAll('.tab-content').forEach(el => el.classList.remove('active'));
    document.querySelectorAll('.tab-btn').forEach(el => el.classList.remove('active'));
    
    document.getElementById(tabId).classList.add('active');
    event.currentTarget.classList.add('active');
}

function actualizarDashboard() {
    let totalVendidas = profes.reduce((acc, p) => acc + p.vendidas, 0);
    let totalRecaudado = totalVendidas * 350; // Tanda actual $350

    document.getElementById("totalEntradas").innerText = `${totalVendidas} / 1.500`;
    document.getElementById("totalRecaudacion").innerText = `$${totalRecaudado.toLocaleString()}`;
    
    let porcentaje = (totalVendidas / 1500) * 100;
    document.getElementById("progressTotal").style.width = `${porcentaje}%`;
}
